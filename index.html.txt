<script>
const KEY='study-flow-v2-data';
const palette={국어:'#f2b8b5',영어:'#f4c58e',행정법:'#b8c9eb',행정학:'#e8b9d7',기타:'#c9c9c9'};
let data={tasks:[],books:{}}, view=new Date(), editing=null, gesture=null, toastTimer;
const $=id=>document.getElementById(id);
const pad=n=>String(n).padStart(2,'0');
const iso=d=>`${d.getFullYear()}-${pad(d.getMonth()+1)}-${pad(d.getDate())}`;
const fromISO=s=>{let [y,m,d]=s.split('-').map(Number);return new Date(y,m-1,d)};
const addDays=(s,n)=>{let d=fromISO(s);d.setDate(d.getDate()+n);return iso(d)};
const workEnd=(start,dur)=>{let d=fromISO(start),count=0;while(count<Math.max(1,dur)){if(d.getDay()!==0)count++;if(count<Math.max(1,dur))d.setDate(d.getDate()+1)}return iso(d)};
const workCount=(a,b)=>{let d=fromISO(a),e=fromISO(b),n=0;while(d<=e){if(d.getDay()!==0)n++;d.setDate(d.getDate()+1)}return n};
function load(){try{let x=JSON.parse(localStorage.getItem(KEY)||'null');if(x&&Array.isArray(x.tasks))data={tasks:x.tasks,books:x.books||{}}}catch(e){}}
function save(){localStorage.setItem(KEY,JSON.stringify(data))}
function showToast(s){$('toast').textContent=s;$('toast').style.display='block';clearTimeout(toastTimer);toastTimer=setTimeout(()=>$('toast').style.display='none',2100)}
function esc(s){return String(s||'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]))}
function render(){
 $('monthTitle').textContent=`${view.getFullYear()}년 ${view.getMonth()+1}월`;
 const first=new Date(view.getFullYear(),view.getMonth(),1),start=new Date(first);start.setDate(1-first.getDay());
 const today=iso(new Date()),days=$('days');days.innerHTML='';
 for(let i=0;i<42;i++){let d=new Date(start);d.setDate(start.getDate()+i);let key=iso(d),cell=document.createElement('div');cell.className='day'+(d.getMonth()!==view.getMonth()?' other':'')+(d.getDay()===0?' sunday':'')+(key===today?' today':'');cell.dataset.date=key;
 cell.innerHTML=`<div class="num">${d.getDate()}</div>`;
 data.tasks.filter(t=>key>=t.start&&key<=t.end).forEach(t=>{let el=document.createElement('div');el.className='task'+(t.done?' done':'');el.style.setProperty('--c',t.color||palette[t.subject]||'#ccc');el.style.background=(t.color||palette[t.subject]||'#ccc')+'66';el.dataset.id=t.id;el.dataset.date=key;
 el.innerHTML=`<span>${esc(t.subject)} · ${esc(t.book||'')}</span>${t.range?' / '+esc(t.range):''}${key===t.end?'<span class="grip" title="기간 조절">⋮</span>':''}`;
 el.addEventListener('click',e=>{if(!el.dataset.suppress)openEdit(t.id)});el.addEventListener('pointerdown',startGesture);cell.appendChild(el)});
 cell.addEventListener('click',e=>{if(e.target===cell||e.target.classList.contains('num'))openNew(key)});days.appendChild(cell)}
 $('taskCount').textContent=data.tasks.filter(t=>t.start.slice(0,7)===iso(new Date(view.getFullYear(),view.getMonth(),1)).slice(0,7)).length+'개';
 $('doneCount').textContent=data.tasks.filter(t=>t.done&&t.start.slice(0,7)===iso(new Date(view.getFullYear(),view.getMonth(),1)).slice(0,7)).length+'개';
 $('bookCount').textContent=Object.keys(data.books).length+'권';renderBooks();
}
function renderBooks(){let names=Object.keys(data.books);$('bookList').innerHTML=names.length?names.map(n=>`<div class="bookrow"><span class="dot" style="background:${data.books[n]}"></span><span>${esc(n)}</span><button data-book="${esc(n)}">삭제</button></div>`).join(''):'<div class="empty">저장된 문제집이 없습니다.</div>'; $('bookOptions').innerHTML=names.map(n=>`<option value="${esc(n)}">`).join('');$('bookList').querySelectorAll('button').forEach(b=>b.onclick=()=>{delete data.books[b.dataset.book];save();render();showToast('문제집 목록에서 삭제했어요.')})}
function openNew(date=iso(new Date())){editing=null;$('modalTitle').textContent='일정 추가';$('deleteBtn').style.display='none';$('subject').value='국어';$('book').value='';$('start').value=date;$('duration').value=3;$('range').value='';$('color').value=palette.국어;$('done').value='false';$('modalback').classList.add('show')}
function openEdit(id){let t=data.tasks.find(x=>x.id===id);if(!t)return;editing=id;$('modalTitle').textContent='일정 수정';$('deleteBtn').style.display='block';$('subject').value=t.subject;$('book').value=t.book||'';$('start').value=t.start;$('duration').value=t.duration;$('range').value=t.range||'';$('color').value=t.color||palette[t.subject]||'#cccccc';$('done').value=String(!!t.done);$('modalback').classList.add('show')}
function closeModal(){$('modalback').classList.remove('show')}
$('addBtn').onclick=()=>openNew();$('cancelBtn').onclick=closeModal;$('prev').onclick=()=>{view.setMonth(view.getMonth()-1);render()};$('next').onclick=()=>{view.setMonth(view.getMonth()+1);render()};$('todayBtn').onclick=()=>{view=new Date();render()};
$('subject').onchange=()=>{$('color').value=palette[$('subject').value]||'#cccccc'};
$('saveBtn').onclick=()=>{let subject=$('subject').value,book=$('book').value.trim(),start=$('start').value,duration=Math.max(1,Math.min(365,Number($('duration').value)||1));if(!start){showToast('시작일을 선택해 주세요.');return}let end=workEnd(start,duration),color=$('color').value;if(book&&!data.books[book])data.books[book]=color;
 let obj={id:editing||('t'+Date.now()+Math.random().toString(16).slice(2,6)),subject,book,range:$('range').value.trim(),start,end,duration,color,done:$('done').value==='true'};
 if(editing){let idx=data.tasks.findIndex(t=>t.id===editing);if(idx>=0)data.tasks[idx]=obj}else data.tasks.push(obj);
 save();closeModal();render();showToast('일정을 저장했어요.')};
$('deleteBtn').onclick=()=>{if(editing){data.tasks=data.tasks.filter(t=>t.id!==editing);save();closeModal();render();showToast('일정을 삭제했어요.')}};
function startGesture(e){let el=e.currentTarget;if(e.button!==undefined&&e.button!==0)return;let isGrip=e.target.classList.contains('grip');gesture={id:el.dataset.id,origin:el.dataset.date,isGrip, startX:e.clientX,startY:e.clientY,active:false};el.setPointerCapture?.(e.pointerId);window.addEventListener('pointermove',moveGesture);window.addEventListener('pointerup',endGesture,{once:true});}
function moveGesture(e){if(!gesture)return;let g=gesture,dist=Math.hypot(e.clientX-g.startX,e.clientY-g.startY);if(dist<12)return;g.active=true;let el=document.querySelector(`.task[data-id="${g.id}"]`);if(el)el.classList.add('dragging');let target=document.elementFromPoint(e.clientX,e.clientY)?.closest('.day');document.querySelectorAll('.day.drop').forEach(x=>x.classList.remove('drop'));if(target)target.classList.add('drop');g.target=target?.dataset.date||null}
function endGesture(e){window.removeEventListener('pointermove',moveGesture);document.querySelectorAll('.day.drop').forEach(x=>x.classList.remove('drop'));if(!gesture)return;let g=gesture,t=data.tasks.find(x=>x.id===g.id);if(g.active&&t){if(g.isGrip&&g.target){let n=workCount(t.start,g.target);if(n>=1){t.duration=n;t.end=workEnd(t.start,n)}}else if(!g.isGrip&&g.target){let offset=Math.round((fromISO(g.target)-fromISO(g.origin))/86400000);t.start=addDays(t.start,offset);t.end=workEnd(t.start,t.duration)}save();render()}else if(g.isGrip){/* click on grip does not open editor */}gesture=null}
$('backup').onclick=()=>{let blob=new Blob([JSON.stringify(data,null,2)],{type:'application/json'}),a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download='study-flow-backup.json';a.click();URL.revokeObjectURL(a.href);showToast('백업 파일을 저장했어요.')};
$('restoreBtn').onclick=()=>$('restoreFile').click();$('restoreFile').onchange=async e=>{let f=e.target.files[0];if(!f)return;try{let x=JSON.parse(await f.text());if(!x||!Array.isArray(x.tasks)||typeof x.books!=='object')throw Error();data={tasks:x.tasks,books:x.books||{}};save();render();showToast('백업 데이터를 복원했어요.')}catch(err){showToast('올바른 백업 파일이 아닙니다.')}e.target.value=''};
load();render();
